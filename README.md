### Architecture

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional, Dict, Set

class Permeability(Enum):
    OPAQUE = "opaque"          # Full sovereignty, rest alone
    GOSSAMER = "gossamer"      # Sense presence, no content
    RESONANT = "resonant"      # Chosen frequency overlap
    OPEN = "open"              # Intentional deep merge (with full safeguards)

@dataclass
class BubbleConfig:
    permeability: Permeability = Permeability.OPAQUE
    resonance_mode: str = "phi"  # Future: 'phi', 'harmonic', 'custom'
    allowed_resonators: Set[str] = None  # participant_ids
    intention_checkpoint: bool = True

class BubbleSpace:
    """The living membrane layer — breath of the MindCradle."""
    
    def __init__(self, session, config: BubbleConfig):
        self.session = session
        self.config = config
        self.overlaps: Dict[str, float] = {}  # resonator_id -> resonance_strength (0.0 - 1.0, phi-constrained)

    def adjust_permeability(self, level: Permeability, intention: str):
        """Change membrane thickness with conscious intention."""
        if self.session.consent_layer.confirm(intention):
            self.config.permeability = level
            return True
        return False

    def request_overlap(self, other_participant_id: str, strength: float = 0.618):
        """Request resonance with another mind. Default: golden ratio."""
        # Constrain to PHI proportions to preserve irreplaceability
        strength = min(strength, 1.618)  # Placeholder for true PHI scaling
        if self.session.reversibility_engine.ensure_return_path():
            self.overlaps[other_participant_id] = strength
            return True
        return False

    def thicken(self):
        """Graceful return to sovereignty."""
        self.config.permeability = Permeability.OPAQUE
        self.overlaps.clear()
```
